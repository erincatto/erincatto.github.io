290B2_INLINE b2Rot b2IntegrateRotation( b2Rot q1, float deltaAngle )
291{
292 // dc/dt = -omega * sin(t)
293 // ds/dt = omega * cos(t)
294 // c2 = c1 - omega * h * s1
295 // s2 = s1 + omega * h * c1
296 b2Rot q2 = { q1.c - deltaAngle * q1.s, q1.s + deltaAngle * q1.c };
297 float mag = sqrtf( q2.s * q2.s + q2.c * q2.c );
298 float invMag = mag > 0.0f ? 1.0f / mag : 0.0f;
299 b2Rot qn = { q2.c * invMag, q2.s * invMag };
300 return qn;
301}

368B2_INLINE float b2ComputeAngularVelocity( b2Rot q1, b2Rot q2, float inv_h )
369{
370 // ds/dt = omega * cos(t)
371 // dc/dt = -omega * sin(t)
372 // s2 = s1 + omega * h * c1
373 // c2 = c1 - omega * h * s1
374
375 // omega * h * s1 = c1 - c2
376 // omega * h * c1 = s2 - s1
377 // omega * h = (c1 - c2) * s1 + (s2 - s1) * c1;
378 // omega * h = s1 * c1 - c2 * s1 + s2 * c1 - s1 * c1
379 // omega * h = s2 * c1 - c2 * s1 = sin(a2 - a1) ~= a2 - a1 for small delta
380 float omega = inv_h * ( q2.s * q1.c - q2.c * q1.s );
381 return omega;
382}

405B2_INLINE b2Rot b2MulRot( b2Rot q, b2Rot r )
406{
407 // [qc -qs] * [rc -rs] = [qc*rc-qs*rs -qc*rs-qs*rc]
408 // [qs qc] [rs rc] [qs*rc+qc*rs -qs*rs+qc*rc]
409 // s(q + r) = qs * rc + qc * rs
410 // c(q + r) = qc * rc - qs * rs
411 b2Rot qr;
412 qr.s = q.s * r.c + q.c * r.s;
413 qr.c = q.c * r.c - q.s * r.s;
414 return qr;
415}
416
419B2_INLINE b2Rot b2InvMulRot( b2Rot a, b2Rot b )
420{
421 // [ ac as] * [bc -bs] = [ac*bc+as*bs -ac*bs+as*bc]
422 // [-as ac] [bs bc] [-as*bc+ac*bs as*bs+ac*bc]
423 // s(b - a) = ac * bs - as * bc
424 // c(b - a) = ac * bc + as * bs
425 b2Rot r;
426 r.s = a.c * b.s - a.s * b.c;
427 r.c = a.c * b.c + a.s * b.s;
428 return r;
429}

432B2_INLINE float b2RelativeAngle( b2Rot a, b2Rot b )
433{
434 // sin(b - a) = bs * ac - bc * as
435 // cos(b - a) = bc * ac + bs * as
436 float s = a.c * b.s - a.s * b.c;
437 float c = a.c * b.c + a.s * b.s;
438 return b2Atan2( s, c );
439}

442B2_INLINE float b2UnwindAngle( float radians )
443{
444 // This function can be implemented using remainderf.
445 // However that is a libc dependency not present in WASM.
446 // This function doesn't work well with huge angles. So unwind your angles regularly.
447 B2_VALIDATE( b2AbsFloat( radians ) < 1.0e4f );
448 float x = b2ClampFloat( radians, -1.0e6f, 1.0e6f );
449
450 // Work in doubles. Adding and removing this constant rounds to the nearest integer.
451 // https://stackoverflow.com/questions/17035464/a-fast-method-to-round-a-double-to-a-32-bit-int-explained
452 double twoPi = 2.0f * B2_PI;
453 double roundToNearest = 6755399441055744.0;
454 double a = x;
455 double k = ( a / twoPi + roundToNearest ) - roundToNearest;
456 return (float)( a - k * twoPi );
457}

630B2_INLINE b2Mat22 b2GetInverse22( b2Mat22 A )
631{
632 float a = A.cx.x, b = A.cy.x, c = A.cx.y, d = A.cy.y;
633 float det = a * d - b * c;
634 if ( det != 0.0f )
635 {
636 det = 1.0f / det;
637 }
638
639 b2Mat22 B = {
640 { det * d, -det * c },
641 { -det * b, det * a },
642 };
643 return B;
644}

648B2_INLINE b2Vec2 b2Solve22( b2Mat22 A, b2Vec2 b )
649{
650 float a11 = A.cx.x, a12 = A.cy.x, a21 = A.cx.y, a22 = A.cy.y;
651 float det = a11 * a22 - a12 * a21;
652 if ( det != 0.0f )
653 {
654 det = 1.0f / det;
655 }
656 b2Vec2 x = { det * ( a22 * b.x - a12 * b.y ), det * ( a11 * b.y - a21 * b.x ) };
657 return x;
658}

706B2_INLINE b2AABB b2MakeAABB( const b2Vec2* points, int count, float radius )
707{
708 B2_ASSERT( count > 0 );
709 b2AABB a = { points[0], points[0] };
710 for ( int i = 1; i < count; ++i )
711 {
712 a.lowerBound = b2Min( a.lowerBound, points[i] );
713 a.upperBound = b2Max( a.upperBound, points[i] );
714 }
715
716 b2Vec2 r = { radius, radius };
717 a.lowerBound = b2Sub( a.lowerBound, r );
718 a.upperBound = b2Add( a.upperBound, r );
719
720 return a;
721}

739 return ( velocity - omega * omegaH * position ) /
740 ( 1.0f + 2.0f * dampingRatio * omegaH + omegaH * omegaH );
741}
